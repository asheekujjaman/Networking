#Basic Settings:
enable
configure terminal
hostname [Switch_Name]
no ip domain-lookup
enable secret [Password]
service password-encryption

#SSH Configure
change hostname
ip domain name hostname+domanin
crypto key generate rsa (ganarate key )
line vty 0 4 
transport input ssh


#Secure Access:
line con 0
password [Password]
login #(login local =>for locol user login)
exit
line vty 0 15
password [Password]
login #(login local =>for locol user login)
transport input ssh
exit

#Create a Local User
configure terminal
username admin secret Admin@123
#Assign Privilege Level (Optional but Recommended)
username admin privilege 15 secret Admin@123


#Management VLAN/IP:
interface vlan 1
ip address [IP_Address] [Subnet_Mask]
no shutdown
exit
ip default-gateway [Gateway_IP]

#VLAN Creation:
vlan [VLAN_ID]
name [VLAN_Name]
exit

#Port Configuration (Access):
interface range [Type] [Interfaces]
switchport mode access
switchport access vlan [VLAN_ID]
spanning-tree portfast
no shutdown
Port Configuration (Trunk/Uplink):
interface [Interface_ID]
switchport mode trunk
switchport trunk allowed vlan [VLAN_IDs]


Log Server:
-----------------
logging host 192.168.1.12 (log server)
logging trap 7 (or other which you want to sent log server)



DHCP Snopping 
-----------------
ip dhcp snooping
ip dhcp snooping vlan 1
interface fastetharnet 0/1 (interface which is connected to dhcp server and allow that vlan 1 or you use)
ip dhcp snooping trust


SNMP Version 1 Configure:
--------------------------
snmp-server community public RW (public=community name)
snmp-server enable traps
snmp-server host 192.168.1.100 version 1 public

SNMP Version 2c Configure:
--------------------------
snmp-server community public RW (public=community name)
snmp-server enable traps
snmp-server host 192.168.1.100 version 2c public

SNMP Version 3 Configure:
--------------------------
snmp-server group SNMPv3-GROUP v3 priv (SNMPv3-GROUP=group name)
snmp-server user snmpuser SNMPv3-GROUP v3 auth sha AuthPass123 priv aes 128 PrivPass123
snmp-server host 192.168.1.100 version 3 priv snmpuser
snmp-server enable traps
